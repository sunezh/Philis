Describe the circuit

import philis as ph

design = ph.Design("inverter")
nmos = design.add_device("nmos", w=1.0, l=0.15)
pmos = design.add_device("pmos", w=2.0, l=0.15)
design.connect(nmos.gate, pmos.gate, name="in")
design.connect(nmos.drain, pmos.drain, name="out")
